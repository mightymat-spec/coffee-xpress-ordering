const APPS_SCRIPT_URL = "https://script.google.com/macros/s/AKfycbyPGbb6wVaFi8ZU3RRC1EPtJyaCypUJ_FM9xq_v4zBzIql-J5jfhyeorWy4gPA1Qu06/exec"; async function loadMenu() { const select = document.getElementById("coffeeType"); const status = document.getElementById("status"); try { const res = await fetch(APPS_SCRIPT_URL).then(r => r.json()); if (res.status === "success" && res.items) { select.innerHTML = ''; res.items.forEach(item => { const opt = document.createElement("option"); opt.value = item.name; opt.textContent = item.price ? `${item.name} ($${item.price})` : item.name; select.appendChild(opt); }); if (status) status.textContent = ""; } else { throw new Error(res.message || "Failed to parse options"); } } catch (err) { console.error(err); if (status) status.textContent = "Failed to load menu options."; } } loadMenu();
