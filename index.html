async function loadMenu() { const select = document.getElementById("coffeeType"); const status = document.getElementById("status"); try { const response = await fetch("https://script.google.com/macros/s/AKfycbyPGbb6wVaFi8ZU3RRC1EPtJyaCypUJ_FM9xq_v4zBzIql-J5jfhyeorWy4gPA1Qu06/exec", { method: "GET", redirect: "follow" }); const res = await response.json(); if (res.status === "success" && res.items) { select.innerHTML = ''; res.items.forEach(item => { const opt = document.createElement("option"); opt.value = item.name; opt.textContent = item.price ? `${item.name} ($${item.price})` : item.name; select.appendChild(opt); }); status.textContent = ""; } else { throw new Error(res.message || "Failed to load menu"); } } catch (err) { console.error("Menu load error:", err); status.className = "status-msg error"; status.textContent = "Failed to load menu options."; } } loadMenu();
