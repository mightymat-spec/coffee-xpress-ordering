`${item.name} ($${item.price})` : item.name; select.appendChild(opt); }); status.textContent = ""; } else { throw new Error(res.message || "Failed to load menu"); } } catch (err) { console.error("Menu load error:", err); status.className = "status-msg error"; status.textContent = "Failed to load menu options."; } } document.getElementById("order-form").addEventListener("submit", async function(e) { e.preventDefault(); const btn = document.getElementById("submit-btn"); const status = document.getElementById("status"); btn.disabled = true; btn.textContent = "Sending Order..."; status.textContent = ""; const orderPayload = { customerName: document.getElementById("customerName").value, coffeeType: document.getElementById("coffeeType").value, size: document.getElementById("size").value, milk: document.getElementById("milk").value, syrup: document.getElementById("syrup").value, notes: document.getElementById("notes").value }; try { const response = await fetch(APPS_SCRIPT_URL, { method: "POST", redirect: "follow", headers: { "Content-Type": "text/plain;charset=utf-8" }, body: JSON.stringify(orderPayload) }); const res = await response.json(); if (res.status === "success") { status.className = "status-msg success"; status.textContent = "✅ Order placed successfully!"; document.getElementById("order-form").reset(); } else { throw new Error(res.message); } } catch (err) { console.error("Submit error:", err); status.className = "status-msg error"; status.textContent = "❌ Failed to place order. Please try again."; } finally { btn.disabled = false; btn.textContent = "Place Order"; } }); loadMenu();
